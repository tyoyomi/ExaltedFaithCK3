ef_actions.1001 = {
	type = character_event
	title = ef_actions.1001.t
	theme = faith

	desc = ef_actions.1001.desc

	# Done first â€” closes hub
	option = {
		name = ef_actions.1001.close

	# Mass Bestow Exalted Faithful
	option = {
		name = ef_actions.1001.mass_bestow
		trigger = {
			NOT = { has_game_rule = ef_rule_granting_disabled }
			custom_tooltip = {
				text = ef_tt_mass_bestow_has_eligible
				OR = {
					ef_has_any_auto_assign_eligible = yes
					any_vassal = { ef_can_receive_exalted_faithful = yes }

			}
		}
		custom_tooltip = ef_mass_bestow_exalted_faithful_decision_effect_tt
		ef_mass_bestow_exalted_faithful_for_patron_effect = yes
		trigger_event = ef_events.1501
	}

	# Auto-Assign Exalted Faithful
	option = {
		name = ef_actions.1001.auto_assign
		trigger = {
			NOT = { has_game_rule = ef_rule_granting_disabled }
			custom_tooltip = {
				text = ef_tt_auto_assign_has_eligible
				ef_has_any_auto_assign_eligible = yes
			}
		}
		custom_tooltip = ef_auto_assign_exalted_faithful_decision_effect_tt
		ef_auto_assign_exalted_faithful_for_patron_effect = yes
		trigger_event = ef_events.1502
	}

	# Strip All Exalted Faithful
	option = {
		name = ef_actions.1001.strip_all
		trigger = {
			has_game_rule = ef_rule_auto_strip_decisions_on

		}
		custom_tooltip = ef_auto_strip_all_exalted_faithful_decision_effect_tt

	option = {
		name = ef_actions.1001.strip_unworthy
		trigger = {
			has_game_rule = ef_rule_auto_strip_decisions_on
			ef_has_any_exalted_faithful = yes
		}
		custom_tooltip = ef_auto_strip_unworthy_exalted_faithful_decision_effect_tt
		ef_auto_strip_unworthy_exalted_faithful_for_patron_effect = yes

	}

	# Convert Realm Faith (next-iter; rehomed off Major)
	option = {
		name = ef_actions.1001.convert
		trigger = {
			has_game_rule = ef_rule_conversion_pulse_on
			NOT = { has_game_rule = ef_rule_granting_disabled }
			OR = {
				has_trait = ef_exalted_faithful
				ef_has_any_exalted_faithful = yes
			}
			custom_tooltip = {
				text = ef_tt_convert_realm_faith_ready
				ef_can_fire_conversion_pulse = yes
			}
			custom_tooltip = {
				text = ef_tt_actions_convert_cd_ready
				NOT = { exists = var:ef_cd_convert_realm }
			}
		}
		custom_tooltip = ef_convert_realm_faith_decision_effect_tt
		ef_convert_realm_faith_effect = yes
		set_variable = {
			name = ef_cd_convert_realm
			value = 1
			days = 365
		}
		trigger_event = ef_events.1601
	}
}
