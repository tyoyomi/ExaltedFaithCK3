ef_manage_exalted_faith_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = major
	desc = ef_manage_exalted_faith_decision_desc
	selection_tooltip = ef_manage_exalted_faith_decision_tooltip
	confirm_text = ef_manage_exalted_faith_decision_confirm

	ai_check_interval = 0

	is_shown = {
		is_alive = yes
		NOT = { has_game_rule = ef_rule_granting_disabled }
		OR = {
			is_ai = no
			has_game_rule = ef_rule_granting_player_and_ai
		}
	}

	is_valid = {
		always = yes
	}

	effect = {
		custom_tooltip = ef_manage_exalted_faith_decision_effect_tt
		trigger_event = ef_settings.1001
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
